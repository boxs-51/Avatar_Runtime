cmake_minimum_required(VERSION 3.22)
project(InstalledAvatarRuntimeConsumer LANGUAGES C CXX)
find_package(AvatarRuntime 0.1 CONFIG REQUIRED COMPONENTS RuntimeCore)
if(NOT AvatarRuntime_PROFILE STREQUAL "CPU_LIFECYCLE_FEATURES_OFF")
    message(FATAL_ERROR "Unexpected SDK profile")
endif()
get_target_property(ars_includes ARS::RuntimeCore INTERFACE_INCLUDE_DIRECTORIES)
foreach(path IN LISTS ars_includes)
    if(NOT EXISTS "${path}/ars/ars.h")
        message(FATAL_ERROR "Missing installed public header")
    endif()
endforeach()
add_executable(installed_c consumer.c)
set_target_properties(installed_c PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO LINKER_LANGUAGE CXX)
add_executable(installed_cpp consumer.cpp)
set_target_properties(installed_cpp PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
foreach(target IN ITEMS installed_c installed_cpp)
    target_link_libraries(${target} PRIVATE ARS::RuntimeCore)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
get_target_property(ars_type ARS::RuntimeCore TYPE)
if(WIN32 AND ars_type STREQUAL "SHARED_LIBRARY")
    foreach(target IN ITEMS installed_c installed_cpp)
        add_custom_command(TARGET ${target} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "$<TARGET_FILE:ARS::RuntimeCore>" "$<TARGET_FILE_DIR:${target}>")
    endforeach()
endif()
enable_testing()
add_test(NAME installed_abi20_c11 COMMAND installed_c)
add_test(NAME installed_abi20_cpp20 COMMAND installed_cpp)
