if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT ARS_ASSET_SANITIZER STREQUAL "ADDRESS")
    message(FATAL_ERROR "Asset fuzz gate requires Clang and real ADDRESS/UBSAN parser instrumentation")
endif()
add_executable(arschar_fuzzer arschar_fuzzer.cpp)
target_link_libraries(arschar_fuzzer PRIVATE ars_asset_components)
ars_asset_options(arschar_fuzzer)
target_compile_options(arschar_fuzzer PRIVATE -fsanitize=fuzzer-no-link)
target_link_options(arschar_fuzzer PRIVATE -fsanitize=fuzzer,address,undefined)
target_compile_definitions(arschar_fuzzer PRIVATE ARS_FUZZ_WORKER="$<TARGET_FILE:ars_schema_worker>")
add_dependencies(arschar_fuzzer ars_schema_worker ars_asset_fixtures)
foreach(target IN ITEMS ars_asset_components ars_asset_zlib ars_asset_png ars_asset_utf8)
    target_compile_options(${target} PRIVATE -fsanitize=fuzzer-no-link)
endforeach()
