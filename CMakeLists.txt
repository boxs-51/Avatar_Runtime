option(ARS_BUILD_ASSET_VALIDATOR "Private CPU validation, no public activation" OFF)
if(ARS_BUILD_ASSET_VALIDATOR)
    set(ARS_ASSET_SANITIZER NONE CACHE STRING "NONE, ADDRESS or THREAD for private parser objects")
    if(NOT ARS_ASSET_SANITIZER MATCHES "^(NONE|ADDRESS|THREAD)$")
        message(FATAL_ERROR "Unknown ARS_ASSET_SANITIZER")
    endif()
    function(ars_asset_options target)
        target_compile_features(${target} PRIVATE cxx_std_20)
        if(MSVC)
            # Private C++ containers never cross the SDK C ABI. MSVC debug proxy
            # allocation inside noexcept allocator constructors cannot propagate
            # a host OOM; use the same container ABI in every private config.
            target_compile_definitions(${target} PRIVATE _ITERATOR_DEBUG_LEVEL=0)
            target_compile_options(${target} PRIVATE /W4 /WX)
            if(ARS_ASSET_SANITIZER STREQUAL "ADDRESS")
                target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
                target_link_options(${target} PRIVATE /INCREMENTAL:NO)
            elseif(ARS_ASSET_SANITIZER STREQUAL "THREAD")
                message(FATAL_ERROR "MSVC TSan is not available; use the Linux worker")
            endif()
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
            if(ARS_ASSET_SANITIZER STREQUAL "ADDRESS")
                target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
                target_link_options(${target} PRIVATE -fsanitize=address,undefined)
            elseif(ARS_ASSET_SANITIZER STREQUAL "THREAD")
                target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
                target_link_options(${target} PRIVATE -fsanitize=thread)
            endif()
        endif()
    endfunction()
    include(cmake/ars_asset_dependencies.cmake)
    find_package(Threads REQUIRED)
    if(WIN32)
        set(worker_platform src/validation/schema_worker_windows.cpp)
    else()
        set(worker_platform src/validation/schema_worker_linux.cpp)
    endif()
    add_library(ars_asset_components STATIC src/assets/owned_bundle.cpp src/assets/zip_reader.cpp src/assets/png_decoder.cpp src/assets/validated_cpu_model.cpp
        src/validation/strict_json.cpp src/validation/manifest_validator.cpp src/validation/profile_b_validator.cpp
        src/validation/schema_worker_client.cpp src/assets/glb_reader.cpp src/validation/glb_validator.cpp ${worker_platform})
