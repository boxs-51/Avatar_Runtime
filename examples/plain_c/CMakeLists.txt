cmake_minimum_required(VERSION 3.22)
project(ArsFoundationExternalC LANGUAGES C)
if(NOT EXISTS "${ARS_FOUNDATION_INFO}")
    message(FATAL_ERROR "Provide ARS_FOUNDATION_INFO from the built foundation tree")
endif()
include("${ARS_FOUNDATION_INFO}")
if(ARS_CONSUMER_RUNTIME_CORE)
    enable_language(CXX)
endif()
if(NOT EXISTS "${ARS_FOUNDATION_LIBRARY}" OR NOT EXISTS "${ARS_FOUNDATION_INCLUDE}/ars/ars.h")
    message(FATAL_ERROR "Missing build-tree foundation library/header")
endif()
add_library(ars_foundation_external UNKNOWN IMPORTED)
set_target_properties(ars_foundation_external PROPERTIES IMPORTED_LOCATION "${ARS_FOUNDATION_LIBRARY}")
add_executable(ars_plain_c main.c)
target_include_directories(ars_plain_c PRIVATE "${ARS_FOUNDATION_INCLUDE}")
target_link_libraries(ars_plain_c PRIVATE ars_foundation_external)
set_target_properties(ars_plain_c PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(ARS_CONSUMER_RUNTIME_CORE)
    set_target_properties(ars_plain_c PROPERTIES LINKER_LANGUAGE CXX)
    target_compile_definitions(ars_plain_c PRIVATE ARS_CONSUMER_RUNTIME_CORE=1)
endif()
if(ARS_FOUNDATION_SHARED)
    target_compile_definitions(ars_plain_c PRIVATE ARS_USE_SHARED=1)
endif()
if(MSVC)
    target_compile_options(ars_plain_c PRIVATE /W4 /WX)
    if(ARS_RUNTIME_SANITIZER STREQUAL "ADDRESS")
        target_compile_options(ars_plain_c PRIVATE /fsanitize=address /Zi)
        target_link_options(ars_plain_c PRIVATE /INCREMENTAL:NO)
    endif()
else()
    target_compile_options(ars_plain_c PRIVATE -Wall -Wextra -Werror)
    if(ARS_FOUNDATION_SANITIZERS)
        target_compile_options(ars_plain_c PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(ars_plain_c PRIVATE -fsanitize=address,undefined)
    endif()
    if(ARS_RUNTIME_SANITIZER STREQUAL "ADDRESS")
        target_compile_options(ars_plain_c PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(ars_plain_c PRIVATE -fsanitize=address,undefined)
    elseif(ARS_RUNTIME_SANITIZER STREQUAL "THREAD")
        target_compile_options(ars_plain_c PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
        target_link_options(ars_plain_c PRIVATE -fsanitize=thread)
    endif()
endif()
enable_testing()
add_test(NAME foundation_external_c COMMAND ars_plain_c)
if(WIN32 AND ARS_FOUNDATION_SHARED)
    set_tests_properties(foundation_external_c PROPERTIES
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${ARS_FOUNDATION_RUNTIME_DIR}")
endif()
