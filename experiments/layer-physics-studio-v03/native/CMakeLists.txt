cmake_minimum_required(VERSION 3.22)
project(LPSNativeMeshCandidate LANGUAGES CXX)
# Isolated native CPU stage: not a replacement for Avatar_Runtime's D3D11 host.
enable_testing()
add_executable(lps_native_mesh_test ../tests/native_mesh_xpbd_tests.cpp)
add_executable(lps_actual_scene_header ../tests/native_real_header.cpp)
add_executable(lps_moc3_overlay_test ../tests/moc3_overlay_tests.cpp)
target_include_directories(lps_actual_scene_header PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../include")
foreach(target IN ITEMS lps_native_mesh_test lps_actual_scene_header lps_moc3_overlay_test)
    target_compile_features(${target} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
add_test(NAME lps_native_mesh_test COMMAND lps_native_mesh_test)
add_test(NAME lps_actual_scene_header COMMAND lps_actual_scene_header)

add_test(NAME lps_moc3_overlay_test COMMAND lps_moc3_overlay_test)
